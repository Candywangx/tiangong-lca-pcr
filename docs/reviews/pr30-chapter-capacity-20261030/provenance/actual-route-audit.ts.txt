import fs from "node:fs";
import path from "node:path";
import assert from "node:assert/strict";
import {createHash} from "node:crypto";
import {gzipSync,gunzipSync} from "node:zlib";
import {parseSiteManifest} from "../packages/pcr-docs/lib/site-contracts.ts";
import {parseGenerationReport} from "../packages/pcr-docs/scripts/generation-report.ts";
import type {SourceBlock} from "../packages/pcr-docs/scripts/markdown.ts";
const [beforeArg,afterArg,hastArg,outArg]=process.argv.slice(2);assert.ok(beforeArg&&afterArg&&hastArg&&outArg);
const before=fs.realpathSync(beforeArg),after=fs.realpathSync(afterArg),hast=fs.realpathSync(hastArg),out=path.resolve(outArg);
assert.ok(!fs.existsSync(out));fs.mkdirSync(out,{recursive:true});
const hash=(bytes:string|Uint8Array)=>"sha256:"+createHash("sha256").update(bytes).digest("hex");
const oldSiteBytes=fs.readFileSync(path.join(before,"site.json")),newSiteBytes=fs.readFileSync(path.join(after,"site.json"));
const oldReportBytes=fs.readFileSync(path.join(before,"report.json")),newReportBytes=fs.readFileSync(path.join(after,"report.json"));
const oldSite=parseSiteManifest(oldSiteBytes),newSite=parseSiteManifest(newSiteBytes),oldReport=parseGenerationReport(oldReportBytes.toString()),newReport=parseGenerationReport(newReportBytes.toString());
assert.equal(oldSite.sourceCommit,"34206f7618a71999e40951d8dc22d2381da3f688");assert.equal(newSite.sourceCommit,"1ca1c00418271410ec0cc2b86805194cd528f03a");
assert.equal(oldReport.sourceCommit,oldSite.sourceCommit);assert.equal(newReport.sourceCommit,newSite.sourceCommit);
assert.deepEqual(oldReport.downloads,newReport.downloads,"every native download URL/name/byte/hash must remain exact");
const omitPages=(record:typeof oldSite.records[number])=>Object.fromEntries(Object.entries(record).filter(([key])=>key!=="pages"));
assert.deepEqual(newSite.records.map(omitPages),oldSite.records.map(omitPages));
const newDocuments=new Map(newReport.documents.map(document=>[document.sourcePath,document]));
assert.equal(newDocuments.size,oldReport.documents.length);assert.equal(newDocuments.size,3092);
const sourceProof=JSON.parse(fs.readFileSync(path.join(hast,"source-groups.json"),"utf8")) as {sourcePath:string;sourceSha256:string;oldParts:number;newParts:number;flattenedHastSha256:string;unchangedPartsMetadata:boolean}[];
const proofs=new Map(sourceProof.map(row=>[row.sourcePath,row]));
const records=[],changedByLanguage:Record<string,number>={},sourceByLanguage:Record<string,number>={};
let removed=0,changed=0,anchorAliases=0,movedAliases=0,oldParts=0,newParts=0,unchanged=0;
for(const document of oldReport.documents){
 const current=newDocuments.get(document.sourcePath);assert.ok(current);assert.equal(current.sourceSha256,document.sourceSha256);
 const original=JSON.parse(fs.readFileSync(path.join(before,document.inventoryPath),"utf8")) as {nodes:SourceBlock[];pages:{url:string;sourceNodeIds:string[]}[];anchorTargets:Record<string,string>};
 const candidate=JSON.parse(fs.readFileSync(path.join(after,current.inventoryPath),"utf8")) as typeof original;
 assert.deepEqual(candidate.nodes,original.nodes);assert.deepEqual(candidate.pages.flatMap(page=>page.sourceNodeIds),original.pages.flatMap(page=>page.sourceNodeIds));
 assert.deepEqual(Object.keys(candidate.anchorTargets).sort(),Object.keys(original.anchorTargets).sort());
 const proof=proofs.get(document.sourcePath);assert.ok(proof);assert.equal(proof.sourceSha256,document.sourceSha256);assert.equal(proof.oldParts,original.pages.length);assert.equal(proof.newParts,candidate.pages.length);
 oldParts+=original.pages.length;newParts+=candidate.pages.length;
 const oldPages=oldSite.pages.filter(page=>page.sourcePath===document.sourcePath).sort((a,b)=>a.part!.index-b.part!.index),newPages=newSite.pages.filter(page=>page.sourcePath===document.sourcePath).sort((a,b)=>a.part!.index-b.part!.index);
 assert.deepEqual(oldPages.map(page=>page.url),original.pages.map(page=>page.url));assert.deepEqual(newPages.map(page=>page.url),candidate.pages.map(page=>page.url));
 const language=oldPages[0]!.language;sourceByLanguage[language]=(sourceByLanguage[language]??0)+1;
 const oldUrls=new Map(original.pages.map((page,index)=>[page.url,index])),newUrls=new Map(candidate.pages.map((page,index)=>[page.url,index]));
 assert.ok(candidate.pages.every(page=>oldUrls.has(page.url)),"no new reading URL may be invented by tail packing");
 const own=new Map(candidate.pages.flatMap((page,index)=>page.sourceNodeIds.map(id=>[id,index] as const)));
 const routing=original.pages.map((page,index)=>{const owners=[...new Set(page.sourceNodeIds.map(id=>own.get(id)))];assert.equal(owners.length,1);assert.ok(owners[0]!==undefined);return{oldPart:index,newPart:owners[0],sourceNodeIds:page.sourceNodeIds};});
 const aliases=Object.entries(original.anchorTargets).map(([anchor,oldTarget])=>{const newTarget=candidate.anchorTargets[anchor];assert.ok(newTarget);const [oldUrl,fragment]=oldTarget.split("#"),[newUrl,newFragment]=newTarget.split("#");assert.equal(fragment,newFragment);assert.ok(oldUrls.has(oldUrl!)&&newUrls.has(newUrl!));return{anchor,oldTarget,newTarget};});
 if(original.pages.length===candidate.pages.length){unchanged++;assert.ok(proof.unchangedPartsMetadata);assert.deepEqual(original.pages,candidate.pages);assert.deepEqual(original.anchorTargets,candidate.anchorTargets);assert.deepEqual(oldPages,newPages);continue;}
 changed++;changedByLanguage[language]=(changedByLanguage[language]??0)+original.pages.length-candidate.pages.length;
 const deleted=original.pages.filter(page=>!newUrls.has(page.url)).map(page=>page.url);removed+=deleted.length;assert.equal(deleted.length,original.pages.length-candidate.pages.length);
 const moved=aliases.filter(row=>row.oldTarget!==row.newTarget);anchorAliases+=aliases.length;movedAliases+=moved.length;
 const key=createHash("sha256").update(document.sourcePath).digest("hex").slice(0,24),fullName="raw/"+key+".json.gz",readableName="routes/"+key+".json";
 const candidateParts=candidate.pages.map((page,index)=>{const body=fs.readFileSync(path.join(after,newPages[index]!.htmlPath!));return{part:index,url:page.url,sourceNodeIds:page.sourceNodeIds,htmlBytes:body.length,htmlSha256:hash(body)};});
 const raw={sourcePath:document.sourcePath,sourceSha256:document.sourceSha256,sourceNodeInventorySha256:hash(JSON.stringify(original.nodes)),flattenedHastSha256:proof.flattenedHastSha256,oldParts:original.pages,partRouting:routing,anchorRouting:aliases,candidateParts};
 const rawBytes=Buffer.from(JSON.stringify(raw,null,2)+"\n"),compressed=gzipSync(rawBytes,{level:9});assert.ok(gunzipSync(compressed).equals(rawBytes));
 fs.mkdirSync(path.join(out,"raw"),{recursive:true});fs.mkdirSync(path.join(out,"routes"),{recursive:true});fs.writeFileSync(path.join(out,fullName),compressed);
 const moves=new Map<string,{fragment:string;aliases:string[];oldPart:number;newPart:number}>();
 for(const row of moved){const [oldUrl,fragment]=row.oldTarget.split("#"),[newUrl]=row.newTarget.split("#"),oldPart=oldUrls.get(oldUrl!)!,newPart=newUrls.get(newUrl!)!;const id=JSON.stringify([fragment,oldPart,newPart]);let move=moves.get(id);if(!move){move={fragment:fragment!,aliases:[],oldPart,newPart};moves.set(id,move);}move.aliases.push(row.anchor);}
 const readable={sourcePath:document.sourcePath,sourceSha256:document.sourceSha256,language,title:oldPages[0]!.title,oldPages:original.pages.map(page=>page.url),newPages:candidate.pages.map(page=>page.url),partRouting:routing.map(({sourceNodeIds,...row})=>({...row,sourceNodes:sourceNodeIds.length})),removedContinuationUrls:deleted,movedAnchors:[...moves.values()],fullEvidence:{path:fullName,originalBytes:rawBytes.length,originalSha256:hash(rawBytes),storedBytes:compressed.length,storedSha256:hash(compressed)}};
 const readableBytes=Buffer.from(JSON.stringify(readable,null,2)+"\n");fs.writeFileSync(path.join(out,readableName),readableBytes);
 records.push({sourcePath:document.sourcePath,sourceSha256:document.sourceSha256,language,title:oldPages[0]!.title,oldPages:original.pages.length,newPages:candidate.pages.length,movedAnchorAliases:moved.length,readable:readableName,readableSha256:hash(readableBytes),...readable.fullEvidence});
}
assert.equal(changed,462);assert.equal(oldParts-newParts,462);assert.equal(unchanged,2630);
for(const record of oldSite.records)for(const language of ["en-US","zh-CN"])assert.ok(newDocuments.has(path.posix.join(record.sourcePath,"pcr."+language+".md")));
assert.ok(fs.readFileSync(path.join(before,"site.json")).equals(oldSiteBytes));assert.ok(fs.readFileSync(path.join(after,"site.json")).equals(newSiteBytes));
const summary={purpose:"Actual canonical before/after route ownership and native-download equality, combined with complete source/HAST comparison; not a replacement for product qualification",baselineSourceCommit:oldSite.sourceCommit,candidateSourceCommit:newSite.sourceCommit,baselineSiteSha256:hash(oldSiteBytes),candidateSiteSha256:hash(newSiteBytes),baselineReportSha256:hash(oldReportBytes),candidateReportSha256:hash(newReportBytes),sourceGroups:oldReport.documents.length,sourceGroupsByLanguage:sourceByLanguage,requiredPcrSources:oldSite.records.length*2,nativeDownloadsUnchanged:oldReport.downloads.length,recordMetadataExceptPageListsExact:true,changedDocuments:changed,unchangedDocuments:unchanged,oldSourceParts:oldParts,newSourceParts:newParts,savedParts:oldParts-newParts,savedPartsByLanguage:changedByLanguage,removedContinuationUrls:removed,newReadingUrls:0,changedDocumentAnchorAliases:anchorAliases,movedAnchorAliases:movedAliases,allOldAnchorsRetained:true,actualRoutesDerivedFromGeneratedInventories:true,records};
fs.writeFileSync(path.join(out,"summary.json"),JSON.stringify(summary,null,2)+"\n");console.log(JSON.stringify({...summary,records:undefined},null,2));
